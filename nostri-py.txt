# I file del backend che divergono da monte, uno per riga, con il path
# relativo alla RADICE del plugin — quindi `py_modules/...` va scritto per
# esteso, e un file come `main.py` si può elencare. Righe vuote e righe che
# iniziano con # sono ignorate.
#
# Questo elenco ha tre lettori, e sbagliarlo non produce un errore ma un
# silenzio: riapplica.sh decide da qui cosa sopravvive al rebase,
# installa-root.sh cosa copiare nel plugin installato, e controlla.sh cosa
# confrontare fra i due. Un file nostro che manca qui torna alla versione di
# monte al primo aggiornamento, senza conflitto e senza una riga di log.
#
# Il nome dice "py" per ragioni storiche: ora ci sono anche un JSON e main.py.

# Rete di sicurezza sulla libreria (issue #405): un fetch fallito non deve
# poter svuotare uno store. sync_run_mixin.py porta anche il rilevatore di
# collasso, tarato sul caso che l'ha reso necessario (603 giochi a zero).
py_modules/unifideck/core/sync_run_mixin.py
py_modules/unifideck/core/sync_service.py
py_modules/unifideck/stores/microsoft/microsoft_catalog.py

# escape_argv per l'avvio fuori dal container Flatpak; monte non ne ha uno.
py_modules/unifideck/launcher/proton/infrastructure/selector.py

# Riespone get_all_playtimes, che monte ha tolto nell'audit §1.2. La nostra
# pagina catalogo ordina per tempo di gioco su tutta la libreria in una volta
# sola; con la sola rotta per-gioco servirebbe una chiamata per titolo.
py_modules/unifideck/rpc/mixins/playtime.py

# La vetrina: giochi gratis e sconti per la home del catalogo. I primi due
# file monte non li ha affatto, quindi non possono entrare in conflitto — ma
# senza una riga qui non verrebbero nemmeno installati. main.py invece è suo,
# e la nostra divergenza sono due righe marcate `NOSTRI in riapplica.sh`:
# l'import del mixin e la sua voce fra le basi di `class Plugin`.
py_modules/unifideck/services/storefront_feed.py
py_modules/unifideck/rpc/mixins/storefront_feed.py
main.py

# Negozio Epic gia' autenticato. web_session.py e' nuovo, quindi non puo'
# entrare in conflitto; storefront.py invece e' di monte, e la divergenza sono
# le ~30 righe di `_signed_in_url_for` piu' la riga che la chiama.
py_modules/unifideck/stores/epic/web_session.py
py_modules/unifideck/launcher/flows/storefront.py
